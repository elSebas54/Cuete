	Void CheckRamps(void)
	{
		Vector3 OldNormal = FloorNormal;
		
		if (player.onground)
		{
			F3DFloor Slomp;
			for(int i = 0; i < FloorSector.Get3DFloorCount(); i++)
			{
				if(FloorSector.Get3DFloor(i).Top.ZAtPoint(Pos.XY) == FloorZ)
				{
					Slomp = FloorSector.Get3DFloor(i);
					break;
				}
			}
			FloorNormal = Slomp ? - Slomp.Top.Normal : FloorSector.FloorPlane.Normal;
			
			Bool TooSteep = Abs(FloorNormal.z) < 0.5;
			Bool Upramp = (Vel + (0,0,Vel.xy dot -OldNormal.xy)) dot FloorNormal > -0.01;//FloorNormal.z > OldNormal.z;
			
			If(FloorNormal != OldNormal)
			{
				Vector3 RampVel;
				If(Upramp) 
				{
					RampVel = OldVel - ( (OldVel dot OldNormal)*OldNormal);
					OldGround = false;
					FloorNormal == (0,0,1);
				}
				else {
					RampVel = OldVel - ( (OldVel dot FloorNormal)*FloorNormal);
					RampVel.z = Min(RampVel.z,0);
				}
				Vel = RampVel;
			}
			else if(!Upramp && FloorNormal.z < 0.9) {
				Vel.xy += FloorNormal.xy*3;
				Vel.z = -FloorNormal.z;
			}
			//If(TooSteep) player.onground = false;
		}
		else If(OldGround) 
		{
			If(FloorNormal != (0,0,1) )
			{
				Vel = OldVel - ( (OldVel dot FloorNormal)*FloorNormal);
				OldGround = false;
			}
			FloorNormal = (0,0,1);
		}
	}

			Angle += (180. / TURN180_TICKS);

		OldGround = Player.OnGround; //used so friction isnt applied in the first tick when landing
		player.onground = (pos.z <= floorz) || bOnMobj || bMBFBouncer || (player.cheats & CF_NOCLIP2);

		//Values for checkin

		Veleng = vel.length();
		VelengXY = vel.xy.length();
		VelAngle = VectorAngle(Vel.X,Vel.Y);
		AirStrafe = VelengXY > MaxSpeed+2 || !(player.onground && OldGround);//Theshold for air strafe velocity, staying below it will apply friction
		
		if (player.playerstate == PST_LIVE && player.cmd.buttons & BT_USE && !(player.oldbuttons & BT_USE) ) 
			A_StartSound("CTaunt",2);

			{
				A_StartSound("*grunt", CHAN_VOICE, CHANF_NORUMBLE);
			}
			else if(vel.z<=0) A_StartSound("*land", 4, 0, 1, 1, 0.5);
		}

	override void MovePlayer ()
	{
		// Player Props and Player Command references 
		let player = self.player;
		UserCmd cmd = player.cmd;
		
		//base maximum move velocity
		MaxSpeed = 5;
		
		if (cmd.forwardmove | cmd.sidemove)
		{
			double MM; // acceletation/vel to add
			double forwardmove, sidemove;
			
			fm = cmd.forwardmove;
			sm = cmd.sidemove; 
			[fm, sm] = TweakSpeeds (fm, sm);
			fm *= Speed / 256; sm *= Speed / 256;
			
			//Set acceleration, cant go faster when trying to move diagonaly
			MM = Min( 50, (fm,sm).Length() ) * 0.032; // * (35 / TICRATE)
			MoveAngle = Angle -90 + VectorAngle(sm,fm);
			VectorMove = (Cos(MoveAngle), Sin(MoveAngle));
			Double projVel = Vel.xy dot VectorMove;//Check how much velocity the is in the direction its trying to move
			
			Player.vel += VectorMove*MM; //this is for view bobbing
			
			If (AirStrafe)// && !player.onground) 
			{
				MaxSpeed *= 0.4;
				MM *= 3;
				If( VelengXY && projVel >= 0 && CVar.GetCVar( "AutoAirStrafe", player ).GetBool() )
				{
					//Does the air strafing for you
					Vector2 Straif = RotateVector(Vel.xy,-90);
					Double StrM = (VectorMove dot Straif) / MM;
					VectorMove = Straif.unit() * Clamp( StrM * 0.5, -1,1);
					projVel = -100;
				}
			}
			else 
			{
				Vector2 Straif = RotateVector(VectorMove,-90).Unit();
				self.vel -= Straif * Clamp( Straif dot Vel.xy, -MM*0.5, MM*0.5);
				// When crouching, speed and bobbing have to be reduced
				if (CanCrouch() && player.crouchfactor != 1)
				{
					MaxSpeed *= 0.5;
				}
			}
			//Increase acceleration when moving against current velocity
			if(player.onground)
			{
				if(projVel < 0.2) 
				{
					MM *= 2;
					player.deltaviewheight -= player.GetMoveBob()*0.5;
				}
			}
			//Limit acceleration to not surpass maxspeed
			MM = Clamp( MaxSpeed - projVel,0,MM);
			//Finally add velocity
			self.vel += VectorMove * MM;
			
			if (!(player.cheats & CF_PREDICTING) && MM != 0)
			{
				PlayRunning ();
			}

			if (player.cheats & CF_REVERTPLEASE)
			{
				player.cheats &= ~CF_REVERTPLEASE;
				player.camera = player.mo;
			}
		}
		else 
		{
			//new friction
			if(!AirStrafe && player.onground && OldGround)
			{
				double calcfric = Max(0, 4*(1 - friction*cursector.friction) );
				If(VelengXY) vel.xy -= vel.xy.unit() * Min(calcfric,VelengXY);
			}
			if(!AirStrafe) PlayIdle();
		}
		//Reduce viewbob when not moving
		Player.vel *= 0.9;
		
		if(player.onground && OldGround)
		{
			if(AirStrafe) 
			{
				Int SpkAmount = Max(1,Veleng*0.1);
				For(Int I; I<SpkAmount; I++)
				{
					VisualThinker spk = VisualThinker.Spawn("NeoSparky", TexMan.CheckForTexture("Models/grem.png"), pos - Vel*I/SpkAmount,
					Vel*1.3 + (fRandom(-2,2),fRandom(-2,2),fRandom(0,2)), 1, 
					SPF_FULLBRIGHT|SPF_ROLL, Random(0,359), (8,8), (0,0), STYLE_ADD );
					NeoSparky(spk).gravity = -0.2*GetGravity();
				}
				A_StartSound("Ballroll",16,CHANF_LOOPING);
				A_SoundVolume(16,Min(0.5,Veleng*0.1) );
				A_SoundPitch(16, Min(4,Veleng*0.05)+0.6 );
			}
			else A_StopSound(16);
		}
		else A_StopSound(16);
	}

	override void CheckJump()
	{
		let player = self.player;
		UserCmd cmd = player.cmd;
		double oldheight = height;
		Double oldfactor = player.crouchfactor;
		Double CrouchHeight = 0.5;
		
		if (player.playerstate != PST_DEAD)//CanCrouch()
		{
			player.crouchdir = player.crouching;//Force player to crouch

			if (player.crouchdir == 0) {
				player.crouchdir = (cmd.buttons & BT_CROUCH) ? -1 : 1;
			}
			//else if (cmd.buttons & BT_CROUCH) player.crouching = 0;
			
			double crouchspeed = player.crouchdir * 0.2;
			player.crouchfactor = clamp(player.crouchfactor + crouchspeed, CrouchHeight, 1);
			
			Double DesHeight = FullHeight * player.crouchfactor;
			If(!player.Onground)
			{
				double zMove = Pos.z + (oldheight - DesHeight)*0.5;
				SetOrigin( (Pos.x,Pos.y,Clamp(zMove, FloorZ, CeilingZ) ) ,true);
			}
			FCheckPosition Col; 
			CheckMove(pos.xy+vel.xy+VectorMove,0,Col);
			//FloorZ = Clamp(Col.FloorZ, FloorZ, Pos.Z+10);
			
			If( Pos.z+DesHeight > Col.CeilingZ && (Pos.z+DesHeight - Col.CeilingZ) <= MaxStepHeight)
			{
				DesHeight = Max( FullHeight*CrouchHeight, Min(Col.CeilingZ,CeilingZ) - Pos.Z);
				player.crouchfactor = DesHeight/FullHeight;
				//If(Col.FloorZ - Pos.z <= MaxStepHeight) SetOrigin( );
			}
			Height = DesHeight;
			Scale.y = Default.Scale.y * player.crouchfactor;
		}
		else
		{
			if (health > 0) Height = FullHeight;
			player.Uncrouch();
		}
		//player.crouchoffset = -player.ViewHeight * (1 - player.crouchfactor);
		
		Double Addjump;
		float maxjump = 9;
		// Check if the player pressed or released the jump button
		if ( player.cmd.buttons & BT_JUMP && player.onground )
		{
			
			double jumpvelz = JumpZ;
			if (player.crouchfactor < 0.8) {jumpvelz *= 0.9;}
			
			Addjump = Clamp( maxjump - vel.z, 0, jumpvelz);
			
			OldGround = false;
			bOnMobj = false;
			FloorNormal = (0,0,1);
			if (!(player.cheats & CF_PREDICTING)) 
			{
				A_StartSound("Cjump", 2);
				A_StartSound("ZipJump", CHAN_BODY, 0, 1, 1, 0.7 + Clamp(Veleng*0.02,0,0.8) );
				//if (InStateSequence(CurState, SeeState) || InStateSequence(CurState, SpawnState) )
				SetStateLabel("Jump");
			}
			
			Vel.Z += Addjump;
		}
		
		double rampup = Pos.Z-OldPosZ;
		//This will add velocity while sliding up ramps or stairs, might make it work better
		If( (OldGround || player.onground) && rampup > 0 && VelengXY > 5) {//OldGround && !player.onground
			Vel.Z = Max(Vel.z, rampup );
			Vel = Vel.unit()*Veleng;
		}
		OldPosZ = Pos.Z;
	}
	
	override void CheckMoveUpDown()
	{
		let player = self.player;
		UserCmd cmd = player.cmd;

		//change upmove with jump and crouch
		if (cmd.buttons & BT_CROUCH) cmd.upmove = -32767;
		if (cmd.buttons & BT_JUMP) cmd.upmove = 32767;
		
		if (cmd.upmove == -32768)
		{ // Only land if in the air
			if (bNoGravity && waterlevel < 2)
			{
				bNoGravity = false;
			}
		}
		else if (cmd.upmove != 0)
		{
			// Clamp the speed to some reasonable maximum.
			cmd.upmove = clamp(cmd.upmove, -0x300, 0x300);
			if (bFly || (player.cheats & CF_NOCLIP2))//waterlevel >= 2 ||  
			{
				Vel.Z += Speed * cmd.upmove * 0.001;
				if (waterlevel < 2 && !bNoGravity) {
					bFly = true;
					bNoGravity = true;
					/*if ((Vel.Z <= -39) && !(player.cheats & CF_PREDICTING))
					{ // Stop falling scream
						A_StopSound(CHAN_VOICE);
					}*/
				}
			}
			else if (cmd.upmove > 0 && !(player.cheats & CF_PREDICTING))
			{
				let fly = FindInventory("ArtiFly");
				if (fly != NULL)
				{
					UseInventory(fly);
				}
			}
			
			If(waterlevel > 1)
			{
				Double MaxVel = cmd.upmove > 0 ? 6 : -6;
				Vel.Z += Clamp(MaxVel - Vel.z, -Speed, Speed);
			}
		}
	}
  
	override bool CanCollideWith(Actor other, bool passive)
	{
		Double Veleng = vel.length();
		If(!Other || (Other.health <= 0 && Other.bISMONSTER) || Other.bCORPSE){Return False;}
		else if ( Veleng > 10 && Other.bSHOOTABLE && (!Other.bNODAMAGE && !Other.bINVULNERABLE) )//
		{
			if( RipperHit(Other) )
			{
				Vector3 Scenter = Pos + (0,0,height*0.5);
				Vector3 OtherDir = Other.pos + (0,0,Other.height*0.5) - Scenter;
				Bool Aliv = Other.Health > 0;
				Int Damg = Clamp(5*Veleng,10,600);
				
				If(Aliv && Other.Mass < 10000 && !Other.bDONTTHRUST) 
				{
					Other.Vel = OtherDir.Unit() * Clamp( (0.3 -other.mass*0.0004 ) * Damg, 0.1, 600);
					If(Other.health <= 0) {}
				}
				Actor Puff = Spawn("RayPuff", Scenter + OtherDir*0.6 );
				Puff.Vel = Other.Vel;
				Other.TriggerPainChance("melee",true);
				Other.DamageMobj(self, self, Damg, "melee", DMG_NO_PAIN|DMG_THRUSTLESS, angle);
				A_StartSound("Weapons/kick",6);
				
				If (!Other || Other.health <= 0) 
				{
					If (Aliv) { 
						A_StartSound("Bing1",21);
						if(other) {Other.Vel += Vel + (0,0,Veleng*0.6); Puff.Vel = Other.Vel*0.9;}
					}
				}
				else {
					OtherDir = OtherDir.Unit();
					Vel += -OtherDir * (OtherDir dot Vel) * 0.5;
				}
			}
			Return False;
		}
		else {Return True;}
	}

	Vector3 WepBob, WepRel, WepVel;
	
	override void Tick()
	{
		OldVel = Vel;
		
		if (player && bWeaponLevel2Ended && !(player.cheats & CF_PREDICTING))
		{
			bWeaponLevel2Ended = false;
			if (player.ReadyWeapon != NULL && player.ReadyWeapon.bPowered_Up)
			{
				player.ReadyWeapon.EndPowerup ();
			}
			if (player.PendingWeapon != NULL && player.PendingWeapon != WP_NOCHANGE &&
				player.PendingWeapon.bPowered_Up &&
				player.PendingWeapon.SisterWeapon != NULL)
			{
				player.PendingWeapon = player.PendingWeapon.SisterWeapon;
			}
		}
		
		bNOFRICTION = true;
		Actor.Tick();
		RipperTick();
		
		vector2 ADir = (cos(pitch), sin(Pitch)*level.pixelstretch);
		double Apitch = asin(Adir.y / ADir.Length() );
		Vector3 Dir = (cos(angle)*cos(Apitch), sin(angle)*cos(Apitch), sin(-Apitch));
		Vector3 YDir = (cos(angle)*cos(Apitch-90), sin(angle)*cos(Apitch-90), sin(-Apitch+90));
		Vector3 XDir = Dir cross YDir;
		
		WepVel += -vel + oldvel*0.8;
		WepVel = WepVel*0.6 + (- WepBob ) * 0.2;
		
		WepBob.x = Clamp(WepBob.x, -20,20);
		WepBob.y = Clamp(WepBob.y, -20,20);
		WepBob.z = Clamp(WepBob.z, -20,20);
		WepBob += WepVel;
		
		WepRel = ( WepBob dot XDir, -WepBob dot YDir, -WepBob dot Dir );
	}
	
	Override void CalcHeight() ///Just added vel.z to viewZ
	{
		let player = self.player;
		double angle;
		double bob;
		bool still = false;

		if (player.cheats & CF_NOCLIP2)
		{
			player.bob = 0;
		}
		else if (bNoGravity && !player.onground)
		{
			player.bob = min(abs(0.5 * FlyBob), MAXBOB);
		}
		else
		{
			player.bob = player.Vel dot player.Vel;
			if (player.bob == 0)
			{
				still = true;
			}
			else
			{
				player.bob *= player.GetMoveBob();

				if (player.bob > MAXBOB)
					player.bob = MAXBOB;
			}
		}

		double defaultviewheight = ViewHeight;

		if (player.cheats & CF_NOVELOCITY)
		{
			player.viewz = pos.Z + defaultviewheight;

			if (player.viewz > ceilingz-0.01)
				player.viewz = ceilingz-0.01;

			return;
		}

		if (bFly && !GetCVar("FViewBob"))
		{
			bob = 0;
		}
		else if (still)
		{
			if (player.health > 0)
			{
				angle = player.BobTimer / (120 * TICRATE / 35.) * 360.;
				bob = player.GetStillBob() * sin(angle);
			}
			else
			{
				bob = 0;
			}
		}
		else
		{
			angle = player.BobTimer / (ViewBobSpeed * TICRATE / 35.) * 360.;
			bob = player.bob * sin(angle) * (waterlevel > 1 ? 0.25f : 0.5f);
		}

		// move viewheight
		if (player.playerstate == PST_LIVE)
		{
			player.viewheight += player.deltaviewheight;

			if (player.viewheight > defaultviewheight)
			{
				player.viewheight = defaultviewheight;
				player.deltaviewheight = 0;
			}
			else if (player.viewheight < (defaultviewheight/2))
			{
				player.viewheight = defaultviewheight/2;
				if (player.deltaviewheight <= 0)
					player.deltaviewheight = 1 / 65536.;
			}

			if (player.deltaviewheight)
			{
				player.deltaviewheight += 2;
				if (!player.deltaviewheight)
					player.deltaviewheight = 1/65536.;
			}
		}

		if (Alternative)
		{
			bob = 0;
		}
		player.viewz = pos.Z + Vel.z + player.viewheight*player.crouchfactor + (bob * clamp(ViewBob, 0. , 1.5)); // [SP] Allow DECORATE changes to view bobbing speed.

		/*if (Floorclip && player.playerstate != PST_DEAD
			&& pos.Z <= floorz)
		{
			player.viewz -= Floorclip;
		}*/
		if (player.viewz > ceilingz - 0.01)
		{
			player.viewz = ceilingz - 0.01;
		}
		if (player.viewz < floorz + 0.01)
		{
			player.viewz = floorz + 0.01;
		}
	}
	
	override Vector2 BobWeapon (double ticfrac)
	{
		//If( Level.IsFrozen() ) Return (0,0);
		
		Vector2 p1, p2, r;
		Vector2 result;

		let player = self.player;
		if (!player) return (0, 0);
		let weapon = player.ReadyWeapon;

		if (weapon == null || weapon.bDontBob)
		{
			return (0, 0);
		}

		for (int i = 0; i < 2; i++)
		{
			// Bob the weapon based on movement speed. ([SP] And user's bob speed setting)
			//double angle = (weapon.BobSpeed * player.GetWBobSpeed() * 35 /	TICRATE*(Level.maptime - 1 + i)) * 0.044;
			double angle = (Level.maptime - 1 + i) * 8 * player.GetWBobSpeed();

			// [RH] Smooth transitions between bobbing and not-bobbing frames.
			// This also fixes the bug where you can "stick" a weapon off-center by
			// shooting it when it's at the peak of its swing.
			if (player.WeaponState & WF_WEAPONBOBBING && curbob != player.bob)
			{
				if (abs(player.bob - curbob) <= 1)
				{
					curbob = player.bob;
				}
				else
				{
					curbob += clamp( (player.bob - curbob)*0.02, -1,1);
				}
			}

			// The weapon bobbing intensity while firing can be adjusted by the player.
			double BobIntensity = (player.WeaponState & WF_WEAPONBOBBING) ? 1. : player.GetWBobFire();

			if (curbob != 0)
			{
				double bobVal = player.bob;
				if (i == 0)
				{
					bobVal = prevBob;
				}
				//[SP] Added in decorate player.viewbob checks * BobIntensity * BobIntensity
				double bobx = (bobVal * weapon.BobRangeX * ViewBob);
				double boby = (bobVal * weapon.BobRangeY * ViewBob);
				switch (weapon.BobStyle)
				{
				case Bob_Normal:
					r.X = bobx * cos(angle);
					r.Y = boby * abs(sin(angle));
					break;

				case Bob_Inverse:
					r.X = bobx*cos(angle);
					r.Y = boby * (1. - abs(sin(angle)));
					break;

				case Bob_Alpha:
					r.X = bobx * sin(angle);
					r.Y = boby * abs(sin(angle));
					break;

				case Bob_InverseAlpha:
					r.X = bobx * sin(angle);
					r.Y = boby * (1. - abs(sin(angle)));
					break;

				case Bob_Smooth:
					r.X = bobx*cos(angle);
					r.Y = boby * ( sin(2*angle) + 0.5);
					//r.Y = 0.5f * (boby * (1. - (cos(angle * 2))));
					break;

				case Bob_InverseSmooth:
					r.X = bobx*cos(angle);
					r.Y = 0.5f * (boby * (1. + (cos(angle * 2))));
				}
			}
			else
			{
				r = (0, 0);
			}
			if (i == 0) p1 = r; else p2 = r;
		}
		
		return p1 * (1. - ticfrac) + p2 * ticfrac + WepRel.xy;
	}
	
	Override Vector3, Vector3 BobWeapon3D (double ticfrac)
	{
		Vector2 oldBob = BobWeapon(ticfrac);
		return ( oldBob.x, -oldBob.y, WepRel.z)*0.5, ( 0, 0, Roll);//( oldBob.x*0.2, -oldBob.y*0.2, Roll)
	}

mixin class NeoRipper //arrays and functions to make a ripper projectile hit an enemy only on the first tick of collision
{
	array<actor> hitactors;
	array<int> hittime;
	
	void RipperTick() //Called every tic
	{
		For(int a; a < hitActors.Size(); a++)
		{
			if(hittime[a]) hittime[a]--;
			else hitActors[a] = null;
		}
	}
	
	bool RipperHit(actor victim, int time = 2) //Called to check if thing should be hit
	{
		Bool washit; 
		int slot = -1;
		For(int a; a<hitActors.Size(); a++)
		{
			If(!hitActors[a]) slot = a;
			If(hitActors[a] == victim) {washit = True; hittime[a] = time;}
		}
		If(!wasHit) 
		{
			If(slot >= 0) {hitActors[slot] = victim; hittime[slot] = time;}
			else {hitActors.Push(victim); hittime.Push(time);}
		}
		Return !wasHit;
	}
}

	static clearscope int GetLineLock( Line l ) 
	{
        int locknum = l.locknumber;
        if ( !locknum ) {
            // check the special
            switch ( l.special ) {
            case FS_Execute:
                locknum = l.Args[2];
                break;
            case Door_LockedRaise:
            case Door_Animated:
                locknum = l.Args[3];
                break;
            case ACS_LockedExecute:
            case ACS_LockedExecuteDoor:
            case Generic_Door:
                locknum = l.Args[4];
                break;
            }
        }
        return locknum;
    }
	
	static bool BreakOpenDoor(Actor Obj, Line Door, Vector3 HitDir)
	{
		Int Lspec = Door.special;
		Obj.A_Logint(lspec);
		
		bool RaiseDoor = (Lspec == 202 ||Lspec == 10|| Lspec == 11|| Lspec == 12|| Lspec == 13|| Lspec == 14 || Lspec == 105||Lspec == 106 || Lspec == 249|| Lspec == 274);
		bool LowerDoor = Lspec == 206 || Lspec == 242;
		
		//check if line is a door that opens moving up and check if you have its key
		If ( (RaiseDoor || LowerDoor) && Obj.CheckKeys(GetLineLock(Door),false,false) && Door.backsector)
		{
			Double DoorMove;
			Double PointZC;
			if(RaiseDoor)
			{
				DoorMove = Door.Frontsector.CeilingPlane.ZatPoint(Door.Frontsector.centerSpot) - 6; //estimation of the height the door opens at
				//DoorMove = Max(DoorMove, Pos.z + height + 20); //making real sure you can pass through
				PointZC = Door.backsector.CeilingPlane.ZatPoint(Door.backsector.centerSpot); //height of the door rn
				
				Door.backsector.MoveCeiling(1000,DoorMove,-1,1,0); //move the door open
			}
			else if(LowerDoor)
			{
				DoorMove = Door.Frontsector.FloorPlane.ZatPoint(Door.Frontsector.centerSpot);
				PointZC = Door.backsector.FloorPlane.ZatPoint(Door.backsector.centerSpot);
				Door.backsector.MoveFloor(1000, Door.Frontsector.FloorPlane.PointToDist(Door.Frontsector.centerSpot,DoorMove) - 6 ,-1,-1,0,1);
			}
			 
			//spawn random smoke around the line
			Vector2 PointA = (Door.Bbox[2],Door.Bbox[1]); //one edge
			Vector2 PointAB = (Door.Bbox[3],Door.Bbox[0]) - PointA; //vector thats the lines length and direction
			//Double PointZF = Door.backsector.FloorPlane.ZatPoint(Door.backsector.centerSpot);
			Double PointZF = DoorMove - PointZC; //how much the door moved
			
			If(Door.delta.x*Door.delta.y < 0) //if it goes the other way tbh i dont remember
			{
				PointA = (Door.Bbox[3],Door.Bbox[1]);
				PointAB = (Door.Bbox[2],Door.Bbox[0]) - PointA; 
			}
			
			float dist = pointAB.Length()*0.2; //how much debree to spawn
			for (int U = 0; U < Dist; U++)
			{
				Float RanPos = Frandom(0,1); //used bellow to get a random position along the line
				actor Debri = Spawn("IceChunk", (PointA.X + RanPos*PointAB.X , PointA.Y + RanPos*PointAB.Y , PointZC + random(0,PointZF) ) );
				Debri.bROLLSPRITE = true;
				Debri.Roll = Random(0,360);
				Debri.bXFLIP = Random(0,1);
				Debri.bYFLIP = Random(0,1);
				Debri.Target = Obj;
				Debri.Vel = HitDir * ( (HitDir dot Obj.Vel) + Random(0,5) ) + ( Frandom(-4,4),Frandom(-4,4),Frandom(-2,4));
			}
			//IceChunk
			Return True;
		}
		else Door.Activate(Obj, 0, SPAC_Use); //else just try using the line
		Return False;
	}

			if(!count) Return;

			for(int i=0; i < count; i++)
			{
				if ((inv = Key(CPlayer.mo.FindInventory(Key.GetKeyType(i)))))
				{
					TextureID icon;

					if (icon.isNull()) icon = inv.Icon; 

					if (icon.isValid()) icons.Push(icon);
				}
			}
			
			int keyscount = icons.Size();
			if(!keyscount) Return;
			double keyAngle = 120*log(keyscount+1);
			for(int i=0; i < keyscount; i++)
			{
				vector2 texsize = TexMan.GetScaledSize(icons[i]) * scale;
				Double kchainfrac = keyAngle/keyscount;
				vector2 circle = actor.AngleToVector( kchainfrac*(i+0.5) - keyAngle*0.5 +90, 4 + 3*keyscount ) * scale;

					DTA_Alpha, 1, DTA_CenterOffset, true,

	override void Draw(int state, double TicFrac)
	{
		Super.Draw (state, TicFrac);
		
		[screenOfsX,screenOfsY,screenSizeX,screenSizeY] = screen.GetViewWindow();
		
		Vector3 wOfs = WobOld + (WobOfs-WobOld)*TicFrac;
		wOfs.Z = 1 + 0.01*wOfs.z;
		
		if (state == HUD_StatusBar || state == HUD_Fullscreen)
		{
			BeginHUD();
	
			DrawString(mHUDFont, "HP:" .. CPlayer.mo.Health, (0, -screenSizeY*0.1) + wOfs.xy, DI_SCREEN_CENTER_BOTTOM|DI_TEXT_ALIGN_CENTER, scale: (1,1)*wOfs.z);
			//CPlayer.mo.GetMaxHealth(true)
			Ammo am1, am2;
			int am1amt, am2amt;
			[am1, am2, am1amt, am2amt] = GetCurrentAmmo();
			String AmmoTxt;
			
			if(am1) AmmoTxt.AppendFormat("%s %i", "AMO:", am1amt);
			if(am2) AmmoTxt.AppendFormat("%s %i", "ALT:", am1amt);
			
			if(AmmoTxt)
			{
				DrawString(mHUDFont, AmmoTxt, (0, -screenSizeY*0.1 + 15*wOfs.Z) + wOfs.xy, DI_SCREEN_CENTER_BOTTOM|DI_TEXT_ALIGN_CENTER, scale: (1,1)*wOfs.z );
			}
			
			DrawString(mHUDFont, String.Format("%s %.2f", "VL:", Cuete(CPlayer.mo).Veleng), (0, -screenSizeY*0.1 + 30*wOfs.Z) + wOfs.xy, DI_SCREEN_CENTER_BOTTOM|DI_TEXT_ALIGN_CENTER, scale: (1,1)*wOfs.z);
			DrawKeys(CPlayer, screenSizeX*0.5 + wOfs.x, screenSizeY*0.78 + wOfs.y + 100*wOfs.z, 2*wOfs.z);
			
			
			If(CVar.GetCVar( "BoostCrosshair", Cplayer ).GetBool()  && CPlayer.readyweapon is "MoveTech")
			{
				double bscl = orscl + (crscl-orscl)*TicFrac;
				String crs = MoveTech(CPlayer.readyweapon).boost ? "LZBMB0" : "LZBMA0";
				
				Screen.DrawTexture(TexMan.CheckForTexture(crs) ,
					false, screenSizeX*0.5, screenSizeY*0.5,
					DTA_ScaleX, bscl, DTA_ScaleY, bscl,
					//DTA_LegacyRenderStyle, STYLE_Shaded,
					//DTA_Color, CrshColor,
					DTA_Rotate, CPlayer.mo.roll );
			}
		}
	}
	
	override void Tick()
	{
		WobOld = WobOfs;
		WobOfs = Cuete(CPlayer.mo).WepRel;
		
		If(CPlayer.readyweapon is "MoveTech" )
		{
			orscl = crscl;
			Double TanFov = Tan(CPlayer.Fov*0.5);
			double bscl = ( 160/tanfov ) / Max( MoveTech(CPlayer.readyweapon).bule.distance,1);
			bscl *= 3;
			crscl = bscl;
		}
		super.Tick();
	}
